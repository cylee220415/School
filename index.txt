<script>
  // 計時器邏輯
  let seconds = 0;
  let timerInterval = setInterval(() => {
    seconds++;
    let mins = Math.floor(seconds / 60);
    let secs = seconds % 60;
    document.getElementById('timer-display').innerText = 
      `時間: ${mins < 10 ? '0' : ''}${mins}:${secs < 10 ? '0' : ''}${secs}`;
  }, 1000);

  // Drag and Drop 功能實作
  let draggedItem = null;

  document.querySelectorAll('.drag-item').forEach(item => {
    item.addEventListener('dragstart', function(e) {
      draggedItem = this;
      setTimeout(() => this.style.display = 'none', 0);
    });

    item.addEventListener('dragend', function() {
      setTimeout(() => {
        draggedItem.style.display = 'inline-block';
        draggedItem = null;
      }, 0);
    });
  });

  document.querySelectorAll('.drop-target, .drag-bank').forEach(target => {
    target.addEventListener('dragover', function(e) {
      e.preventDefault();
      this.classList.add('hover');
    });

    target.addEventListener('dragleave', function() {
      this.classList.remove('hover');
    });

    target.addEventListener('drop', function(e) {
      e.preventDefault();
      this.classList.remove('hover');
      if (draggedItem) {
        // 如果放置點已經有元素，將原本的放回對應的 bank
        if (this.classList.contains('drop-target') && this.children.length > 0) {
          let parentBank = draggedItem.parentElement;
          parentBank.appendChild(this.children[0]);
        }
        this.appendChild(draggedItem);
      }
    });
  });

  // 測驗提交與批改
  function submitQuiz() {
    const sClass = document.getElementById('student-class').value;
    const sNo = document.getElementById('student-number').value;
    const sName = document.getElementById('student-name').value;

    if (!sClass || !sNo || !sName) {
      alert('請先完整輸入【班級】、【座號】與【姓名】再提交！');
      return;
    }

    clearInterval(timerInterval); // 停止計時

    let totalQuestions = 0;
    let correctQuestions = 0;

    // 檢查答案
    document.querySelectorAll('.drop-target').forEach(target => {
      totalQuestions++;
      const expected = target.getAttribute('data-ans');
      const child = target.querySelector('.drag-item');
      const feedback = target.parentElement.querySelector('.answer-feedback') || target.nextElementSibling;

      if (child && child.getAttribute('data-val') === expected) {
        correctQuestions++;
        target.style.borderColor = 'var(--success)';
        if (feedback) {
          feedback.innerText = '✓ 正確';
          feedback.className = 'answer-feedback correct';
          feedback.style.display = 'inline';
        }
      } else {
        target.style.borderColor = 'var(--danger)';
        if (feedback) {
          feedback.innerText = `✗ 錯誤 (正確解答: ${expected})`;
          feedback.className = 'answer-feedback incorrect';
          feedback.style.display = 'inline';
        }
      }
    });

    // 顯示成果
    const score = Math.round((correctQuestions / totalQuestions) * 100);
    const resultBoard = document.getElementById('result-board');
    document.getElementById('score-text').innerText = `${sClass}班 ${sNo}號 ${sName} - 得分：${score} 分 (${correctQuestions}/${totalQuestions})`;
    document.getElementById('time-text').innerText = `總共耗時：${document.getElementById('timer-display').innerText.replace('時間: ', '')}`;
    resultBoard.style.display = 'block';

    window.scrollTo({ top: 0, behavior: 'smooth' });
  }
</script>
